# Implementation Plan: Service History + Smart Notifications

**Branch**: `007-service-history-notifications` | **Date**: 2026-05-25 | **Spec**: [spec.md](spec.md)

## Summary

Add service-history logging (`car_logs`), Firebase Cloud Messaging push notifications, a gas-station check-in endpoint with quick fill-up, and admin-managed fuel prices to the car tracker API. The feature introduces three new domain entities (`CarLog`, `DeviceToken`, `FuelPrice`), a Firebase infrastructure layer ported from tai-be-superapp, three queued event/listener pairs, two scheduled Artisan commands, and a migration delta on `fill_ups` that adds `fuel_type`, nullable `liters`, `station_lat`, and `station_lng`.

```php
    $table->id();
    $table->foreignId('car_id')->nullable()->constrained('cars')->nullOnDelete();
    $table->foreignId('service_id')->nullable()->constrained('services')->nullOnDelete();
    $table->integer('odometer_at_service');
    $table->decimal('actual_cost', 10, 2);
    $table->date('performed_at');
    $table->timestamps();

    $table->index('car_id');
    $table->index('service_id');
});

Schema::create('device_tokens', function (Blueprint $table) {
    $table->id();
    $table->foreignId('user_id')->nullable()->constrained('users')->nullOnDelete();
    $table->string('token');
    $table->enum('device', ['android', 'ios']);
    $table->timestamps();

    $table->unique(['user_id', 'device']);
});

Schema::create('fuel_prices', function (Blueprint $table) {
    $table->id();
    $table->enum('type', ['92', '95', 'electric']);
    $table->decimal('price_per_unit', 10, 2);
    $table->date('effective_from');
    $table->timestamps();

    $table->index(['type', 'effective_from']);

    $table->enum('fuel_type', ['92', '95', 'electric'])->nullable()->after('fill_date');
```

### Repository Spatie property set

```php
// CarLogRepositoryEloquent
protected array $allowedIncludes     = ['car', 'service'];
protected array $allowedFilters      = [];
protected array $allowedFiltersExact = ['car_id', 'service_id'];
protected array $allowedSorts        = ['performed_at', 'actual_cost', 'odometer_at_service'];
protected array $allowedDefaultSorts = ['-performed_at'];

protected function scopeToUser(): void
{
    if (auth()->check() && !auth()->user()->isAdmin()) {
        $this->model = $this->model->whereHas('car', function ($q) {
            $q->where('user_id', auth()->id());
        });
    }
}

// FuelPriceRepositoryEloquent
protected array $allowedIncludes     = [];
protected array $allowedFilters      = [];
protected array $allowedFiltersExact = ['type'];
protected array $allowedSorts        = ['effective_from', 'type'];
protected array $allowedDefaultSorts = ['-effective_from'];

public function currentForType(string $type): ?FuelPrice
{
    return app($this->model())->newQuery()
        ->where('type', $type)
        ->orderByDesc('effective_from')
        ->first();
}

// DeviceTokenRepository contract (custom method only)
interface DeviceTokenRepository extends RepositoryInterface
{
    public function upsertToken(int $userId, string $token, string $device): void;
}

// DeviceTokenRepositoryEloquent implementation
public function upsertToken(int $userId, string $token, string $device): void
{
    app($this->model())->newQuery()->updateOrCreate(
        ['user_id' => $userId, 'device' => $device],
        ['token' => $token]
    );
    $this->resetModel();
}
```

```php
{
    if (auth()->check() && $request->hasHeader('DEVICE-TOKEN') && $request->hasHeader('DEVICE-TYPE')) {
```

### Controllers

```php
// CarLogController — full CRUD
public function index(Request $request, Car $car): JsonResponse
{
    $this->authorize('viewAny', [CarLog::class, $car]);
    $logs = $this->carLogRepository->where('car_id', $car->id)->spatie()->paginate();
    return $this->paginated($logs, CarLogResource::class);
}

public function store(StoreCarLogRequest $request, Car $car): JsonResponse
{
    try { DB::beginTransaction();
        $log = $this->carLogRepository->create([...$request->validated(), 'car_id' => $car->id]);
        event(new CarLogCreated($log));
        DB::commit();
        return $this->success(new CarLogResource($log), 201, 'Service log created.');
    } catch (\Exception $e) { DB::rollBack(); return $this->error($e->getMessage(), 422); }
}

public function show(Request $request, Car $car, CarLog $log): JsonResponse
{
    $this->authorize('view', $log);
    abort_if($log->car_id !== $car->id, 404);
    return $this->success(new CarLogResource($log));
}

public function update(UpdateCarLogRequest $request, Car $car, CarLog $log): JsonResponse
{
    abort_if($log->car_id !== $car->id, 404);
    try { DB::beginTransaction();
        $updated = $this->carLogRepository->update($request->validated(), $log->id);
        DB::commit();
        return $this->success(new CarLogResource($updated));
    } catch (\Exception $e) { DB::rollBack(); return $this->error($e->getMessage(), 422); }
}

public function destroy(Request $request, Car $car, CarLog $log): JsonResponse
{
    $this->authorize('delete', $log);
    abort_if($log->car_id !== $car->id, 404);
    $this->carLogRepository->delete($log->id);
    return $this->success([], 200, 'Service log deleted.');
}

// GasStationCheckInController — invokable
public function __invoke(Request $request, Car $car): JsonResponse
{
    $this->authorize('create', [FillUp::class, $car]); // car owner gate
    event(new GasStationCheckIn($car));
    return $this->success([], 200, 'Notification sent.');
}

// FillUpController::quick (new action)
public function quick(QuickFillUpRequest $request, Car $car): JsonResponse
{
    try { DB::beginTransaction();
        $liters = $request->liters;
        if ($liters === null) {
            $price = $this->fuelPriceRepository->currentForType($request->fuel_type);
            $liters = $price ? round($request->amount_paid / $price->price_per_unit, 2) : null;
        }
        $fillUp = $this->fillUpRepository->create([
            'car_id'      => $car->id,
            'liters'      => $liters,
            'odometer'    => $car->current_km,
            'cost_egp'    => $request->amount_paid,
            'fill_date'   => now()->toDateString(),
            'fuel_type'   => $request->fuel_type,
            'station_lat' => $request->station_lat,
            'station_lng' => $request->station_lng,
        ]);
        DB::commit();
        return $this->success(new FillUpResource($fillUp), 201, 'Fill-up recorded successfully.');
    } catch (\Exception $e) { DB::rollBack(); return $this->error($e->getMessage(), 422); }
}

// FuelPriceController
public function index(Request $request): JsonResponse
{
    $this->authorize('viewAny', FuelPrice::class);
    $prices = $this->fuelPriceRepository->spatie()->paginate();
    return $this->paginated($prices, FuelPriceResource::class);
}

public function store(StoreFuelPriceRequest $request): JsonResponse
{
    try { DB::beginTransaction();
        $price = $this->fuelPriceRepository->create($request->validated());
        DB::commit();
        return $this->success(new FuelPriceResource($price), 201, 'Fuel price created.');
    } catch (\Exception $e) { DB::rollBack(); return $this->error($e->getMessage(), 422); }
}

public function update(UpdateFuelPriceRequest $request, FuelPrice $fuelPrice): JsonResponse
{
    try { DB::beginTransaction();
        $updated = $this->fuelPriceRepository->update($request->validated(), $fuelPrice->id);
        DB::commit();
        return $this->success(new FuelPriceResource($updated));
    } catch (\Exception $e) { DB::rollBack(); return $this->error($e->getMessage(), 422); }
}
```
